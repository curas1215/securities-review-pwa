  function enhanceHtml(original) {
    let html = original;
    const viewport = '<meta name="viewport" content="width=device-width,initial-scale=1,viewport-fit=cover">';
    const additions = viewport + enhancementCss();
    if (/<\/head>/i.test(html)) html = html.replace(/<\/head>/i, `${additions}</head>`);
    else html = additions + html;
    const script = enhancementScript();
    if (/<\/body>/i.test(html)) html = html.replace(/<\/body>/i, `${script}</body>`);
    else html += script;
    return html;
  }

  async function openStudy(restore = false) {
    const baseline = await idbGet('baseline', BASELINE_ID);
    if (!baseline || !baseline.blob) { toast('请先导入原始复习 HTML'); return; }
    els.launcher.classList.add('hidden');
    els.studyView.classList.remove('hidden');
    els.studyStatus.textContent = `原数据只读 · ${baseline.sha256.slice(0, 8)}…`;
    const original = await baseline.blob.text();
    const enhanced = enhanceHtml(original);
    if (frameBlobUrl) URL.revokeObjectURL(frameBlobUrl);
    frameBlobUrl = URL.createObjectURL(new Blob([enhanced], {type:'text/html;charset=utf-8'}));
    restoreAfterFrameLoad = restore ? await idbGet('settings','lastPosition') : null;
    els.frame.src = frameBlobUrl;
  }

  function closeStudy() {
    els.studyView.classList.add('hidden');
    els.launcher.classList.remove('hidden');
    els.frame.src = 'about:blank';
    if (frameBlobUrl) { URL.revokeObjectURL(frameBlobUrl); frameBlobUrl = null; }
    updateDashboard();
  }

  async function handleAttempt(msg) {
    const id = msg.qid ? `qid:${msg.qid}` : `stem:${await textKey(msg.stem)}`;
    const existing = await idbGet('attempts', id) || { id, qid: msg.qid || '', stem: msg.stem || '', attempts: 0, correctCount: 0, wrongCount: 0, history: [] };
    existing.stem = msg.stem || existing.stem;
    existing.qid = msg.qid || existing.qid;
    existing.attempts += 1;
    if (msg.isCorrect) existing.correctCount += 1; else existing.wrongCount += 1;
    existing.lastSelected = msg.selected;
    existing.correctAnswer = msg.correct;
    existing.lastCorrect = Boolean(msg.isCorrect);
    existing.lastAt = msg.at || new Date().toISOString();
    existing.history = (existing.history || []).concat([{selected:msg.selected,correct:msg.correct,isCorrect:Boolean(msg.isCorrect),at:existing.lastAt}]).slice(-10);
    await idbPut('attempts', existing);
    await updateDashboard();
    if (!msg.isCorrect) toast('已自动加入错题记录');
  }

  async function handleFavorite(msg) {
    const label = String(msg.label || '').trim();
    if (!label) return;
    const id = `fav:${await textKey(label)}`;
    if (msg.active) await idbPut('favorites', { id, label, at: msg.at || new Date().toISOString() });
    else await idbDelete('favorites', id);
    await updateDashboard();
  }

  async function handlePosition(msg) {
    if (!msg.label && !msg.tab) return;
    const value = { key:'lastPosition', tab:msg.tab || '', label:msg.label || '', at:msg.at || new Date().toISOString() };
    await idbPut('settings', value);
    els.lastPosition.textContent = value.label ? value.label.slice(0, 22) : (value.tab || '已记录');
  }

  async function updateDashboard() {
    const [attempts, favorites, last] = await Promise.all([idbAll('attempts'), idbAll('favorites'), idbGet('settings','lastPosition')]);
    const wrong = attempts.filter(x => (x.wrongCount || 0) > 0);
    els.wrongCount.textContent = String(wrong.length);
    els.favoriteCount.textContent = String(favorites.length);
    els.lastPosition.textContent = last ? ((last.label || last.tab || '已记录').slice(0, 22)) : '尚无记录';
  }
