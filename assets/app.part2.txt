    const hash = await sha256(buffer);
    if (expectedHash && hash !== expectedHash) throw new Error('内置基线 SHA-256 与配置不一致，已阻止导入。');
    const blob = new Blob([buffer], { type: 'text/html;charset=utf-8' });
    const text = await blob.text();
    const validation = baselineChecks(text, filename);
    if (!validation.ok) throw new Error('文件不像预期的证券分析师复习 HTML：未检测到必要的考纲/题库结构。');
    const record = {
      id: BASELINE_ID,
      filename,
      size: buffer.byteLength,
      sha256: hash,
      importedAt: new Date().toISOString(),
      source,
      validation: validation.checks,
      blob
    };
    await idbPut('baseline', record);
    currentBaseline = record;
    const persistent = await requestPersistentStorage();
    await renderBaseline();
    toast(persistent ? '原始题库已锁定保存；已申请持久存储' : '原始题库已锁定保存');
    return record;
  }

  async function importBaselineFile(file) {
    if (!file) return;
    els.importHint.textContent = '正在读取并校验原文件…';
    try {
      const buffer = await file.arrayBuffer();
      const record = await importBaselineArrayBuffer(buffer, file.name, null, 'manual');
      const filenameNote = file.name === EXPECTED_FILENAME ? '' : '（文件名与推荐基线名不同，但内容结构校验通过）';
      els.importHint.textContent = `导入成功 ${filenameNote} SHA-256: ${record.sha256.slice(0, 16)}…`;
    } catch (err) {
      els.importHint.textContent = `导入失败：${err.message}`;
      toast(`导入失败：${err.message}`);
    } finally {
      els.baselineFile.value = '';
      els.replaceBaselineFile.value = '';
    }
  }

  async function maybeImportEmbedded() {
    const existing = await idbGet('baseline', BASELINE_ID);
    if (existing) return;
    const cfg = await loadEmbeddedConfig();
    if (!cfg || !cfg.embedded) return;
    try {
      const res = await fetch(cfg.path || 'baseline/original.html');
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const buffer = await res.arrayBuffer();
      await importBaselineArrayBuffer(buffer, cfg.filename || EXPECTED_FILENAME, cfg.sha256 || null, 'embedded');
    } catch (err) {
      toast(`内置题库载入失败：${err.message}`);
    }
  }

  async function renderBaseline() {
    currentBaseline = await idbGet('baseline', BASELINE_ID);
    const has = Boolean(currentBaseline);
    els.noBaseline.classList.toggle('hidden', has);
    els.hasBaseline.classList.toggle('hidden', !has);
    els.removeBaselineBtn.classList.toggle('hidden', !has);
    if (has) {
      els.baselineName.textContent = currentBaseline.filename;
      els.baselineSize.textContent = formatBytes(currentBaseline.size);
      els.baselineHash.textContent = currentBaseline.sha256;
      els.baselineImportedAt.textContent = formatDate(currentBaseline.importedAt);
      const v = currentBaseline.validation || {};
      els.baselineIntegrity.textContent = (v.title && v.syllabus && v.questions) ? '结构校验通过 · 基线只读' : '已保存 · 建议重新校验';
    }
    await updateDashboard();
  }
